cmake_minimum_required(VERSION 3.16)
project(RicattiSNS LANGUAGES CXX)
add_library(sns Keldysh/sns/spectral.cpp Keldysh/sns/kinetic.cpp Keldysh/sns/current.cpp Keldysh/sns/runner.cpp)
target_include_directories(sns PUBLIC Keldysh/sns)
target_compile_features(sns PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(sns PRIVATE /W4 /utf-8)
else()
  target_compile_options(sns PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(sns_cli Keldysh/sns/cli.cpp)
target_link_libraries(sns_cli PRIVATE sns)
include(CTest)
if(BUILD_TESTING)
  add_executable(sns_tests Keldysh/sns/tests.cpp)
  target_link_libraries(sns_tests PRIVATE sns)
  add_test(NAME sns_unit_and_sanity COMMAND sns_tests)
endif()
